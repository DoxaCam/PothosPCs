-- =========================

DROP TABLE IF EXISTS ServiceEmployee;
DROP TABLE IF EXISTS ServicePart;
DROP TABLE IF EXISTS Service;
DROP TABLE IF EXISTS PC;
DROP TABLE IF EXISTS Orders;
DROP TABLE IF EXISTS Part;
DROP TABLE IF EXISTS Employee;
DROP TABLE IF EXISTS Supplier;
DROP TABLE IF EXISTS Customer;
-- =========================

-- 1. Independent Tables
CREATE TABLE Customer (
    customer_ID INT AUTO_INCREMENT PRIMARY KEY,
    email VARCHAR(255) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    first_Name VARCHAR(100) NOT NULL,
    last_Name VARCHAR(100) NOT NULL,
    phone VARCHAR(20) NOT NULL
);

CREATE TABLE Supplier (
    supplier_ID INT AUTO_INCREMENT PRIMARY KEY,
    supplier_name VARCHAR(150) NOT NULL,
    contact_info VARCHAR(150) NOT NULL,
    supplier_addr TEXT NOT NULL
);

CREATE TABLE Employee (
    emp_ID INT AUTO_INCREMENT PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    role VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL UNIQUE
);

-- 2. First-Level Dependent Tables
CREATE TABLE Orders (
    order_ID INT AUTO_INCREMENT PRIMARY KEY,
    order_date DATE NOT NULL,
    total_price DECIMAL(10,2) NOT NULL,
    payment_method VARCHAR(50) NOT NULL,
    customer_ID INT NOT NULL,
    FOREIGN KEY (customer_ID) REFERENCES Customer(customer_ID)
		ON DELETE RESTRICT
		ON UPDATE CASCADE
);

CREATE TABLE Part (
    part_ID INT AUTO_INCREMENT PRIMARY KEY,
    part_name VARCHAR(150) NOT NULL,
    category VARCHAR(100) NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    availability_status VARCHAR(50) NOT NULL,
    eta DATE,
    supplier_ID INT NOT NULL,
    FOREIGN KEY (supplier_ID) REFERENCES Supplier(supplier_ID)
		ON DELETE RESTRICT
		ON UPDATE CASCADE
);

-- 3. Second-Level Dependent Tables
CREATE TABLE PC (
    pc_ID INT AUTO_INCREMENT PRIMARY KEY,
    model_name VARCHAR(150) NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    specs TEXT NOT NULL,
    type VARCHAR(50) NOT NULL, 
    order_ID INT,
    FOREIGN KEY (order_ID) REFERENCES Orders(order_ID)
		ON DELETE SET NULL
		ON UPDATE CASCADE
);

-- 4. Third-Level Dependent Tables
CREATE TABLE Service (
    service_ID INT AUTO_INCREMENT PRIMARY KEY,
    service_type VARCHAR(50) NOT NULL, 
    eta DATE NOT NULL,
    labor_cost DECIMAL(10,2) NOT NULL,
    status VARCHAR(50) NOT NULL,
    pc_ID INT NOT NULL,
    FOREIGN KEY (pc_ID) REFERENCES PC(pc_ID)
		ON DELETE RESTRICT
		ON UPDATE CASCADE
);

-- 5. Junction Tables (Many-to-Many Relationships)
CREATE TABLE ServiceEmployee (
    service_ID INT NOT NULL,
    emp_ID INT NOT NULL,
    PRIMARY KEY (service_ID, emp_ID),
    FOREIGN KEY (service_ID) REFERENCES Service(service_ID)
		ON DELETE CASCADE
		ON UPDATE CASCADE,
    FOREIGN KEY (emp_ID) REFERENCES Employee(emp_ID)
		ON DELETE CASCADE
		ON UPDATE CASCADE
);

CREATE TABLE ServicePart (
    service_ID INT NOT NULL,
    part_ID INT NOT NULL,
    PRIMARY KEY (service_ID, part_ID),
    FOREIGN KEY (service_ID) REFERENCES Service(service_ID)
		ON DELETE CASCADE
		ON UPDATE CASCADE,
    FOREIGN KEY (part_ID) REFERENCES Part(part_ID)
		ON DELETE CASCADE
		ON UPDATE CASCADE
);
